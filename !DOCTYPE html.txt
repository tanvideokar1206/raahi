        function renderCards(data) {
            const grid = document.getElementById('placesGrid');
            grid.innerHTML = '';

            if (data.length === 0) {
                grid.innerHTML = '<p style="text-align:center; grid-column: 1/-1; color: var(--text-muted);">No cultural items found matching your search.</p>';
                return;
            }

            data.forEach(item => {
                const card = document.createElement('div');
                card.className = 'card';
                card.innerHTML = `
                    <div class="card-header">
                        <span>${item.state}</span>
                        <h3>${item.title}</h3>
                    </div>
                    <div class="card-body">
                        <p><strong>Culture & History:</strong> ${item.history}</p>
                        <p><strong>Iconic Destination:</strong> ${item.place}</p>
                        <div style="margin-top: 1rem;">
                            <span class="tag">${item.specialty}</span>
                        </div>
                    </div>
                `;
                grid.appendChild(card);
            });
        }

        function filterPlaces() {
            const query = document.getElementById('searchInput').value.toLowerCase();
            const filtered = culturalData.filter(item => 
                item.state.toLowerCase().includes(query) ||
                item.title.toLowerCase().includes(query) ||
                item.history.toLowerCase().includes(query) ||
                item.place.toLowerCase().includes(query) ||
                item.specialty.toLowerCase().includes(query)
            );
            renderCards(filtered);
        }

        // Initial render on page load
        renderCards(culturalData);
